function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = sheet.getDataRange().getValues(); const headers = data[0]; const rows = data.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }).filter(r => r['Date'] !== '' && r['Date'] !== undefined); return ContentService .createTextOutput(JSON.stringify(rows)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const payload = JSON.parse(e.postData.contents); if (payload.action === 'save' && Array.isArray(payload.trades)) { const lastRow = sheet.getLastRow(); if (lastRow > 1) sheet.deleteRows(2, lastRow - 1); payload.trades.forEach(t => { sheet.appendRow([ t.date, t.pair, t.dir, t.result, t.entry || '', t.exit || '', t.lots || '', t.comm || 0, t.pnl || 0, t.rr || 0, t.setup || '', t.session || '', t.tf || '', t.notes || '', t.id || '', t.asset || '' ]); }); } return ContentService .createTextOutput(JSON.stringify({status:'ok'})) .setMimeType(ContentService.MimeType.JSON); }
